cmake_minimum_required(VERSION 3.19)
# 独立子工程；不导入上层 CMake，不向旧 target 传播寄存器上限。
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 89 CACHE STRING "CUDA architectures")
endif()
project(VitaminDbfRegisterExperiment LANGUAGES CXX CUDA)
find_package(CUDAToolkit 12.4 REQUIRED)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
get_filename_component(SGEMM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
foreach(variant IN ITEMS default maxr128)
  set(target "windows_dbf_${variant}_benchmark")
  add_executable(${target} "${SGEMM_ROOT}/scripts/windows_dbf_register_benchmark.cu")
  target_link_libraries(${target} PRIVATE CUDA::cublas)
  target_compile_options(${target} PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:-O3;-lineinfo;-Xptxas=-v;--expt-relaxed-constexpr;--expt-extended-lambda>)
  if(MSVC)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8>)
    target_compile_definitions(${target} PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
  endif()
  if(variant STREQUAL "maxr128")
    target_compile_definitions(${target} PRIVATE DBF_MAX_REGISTERS=128)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--maxrregcount=128>)
  else()
    target_compile_definitions(${target} PRIVATE DBF_MAX_REGISTERS=0)
  endif()
endforeach()
