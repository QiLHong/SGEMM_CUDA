cmake_minimum_required(VERSION 3.19)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 120 CACHE STRING "CUDA architectures")
endif()
project(RTX5090_STRICT_FP32 LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_executable(rtx5090_strict_fp32 benchmark.cu)
set_target_properties(rtx5090_strict_fp32 PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
target_compile_definitions(rtx5090_strict_fp32 PRIVATE SGEMM_K10_MIN_BLOCKS=2)
target_compile_options(rtx5090_strict_fp32 PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo;-Xptxas=-v>)
target_link_libraries(rtx5090_strict_fp32 PRIVATE CUDA::cublas)
