cmake_minimum_required(VERSION 3.19)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 89 CACHE STRING "CUDA architectures")
endif()
project(TensorCoreGridLearning LANGUAGES CXX CUDA)
find_package(CUDAToolkit 12.4 REQUIRED)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
add_executable(windows_tensor_grid_benchmark benchmark.cu)
target_link_libraries(windows_tensor_grid_benchmark PRIVATE CUDA::cublas)
target_compile_options(windows_tensor_grid_benchmark PRIVATE
  $<$<COMPILE_LANGUAGE:CUDA>:-O3;-lineinfo;-Xptxas=-v;--expt-relaxed-constexpr;--expt-extended-lambda>)
if(MSVC)
  target_compile_options(windows_tensor_grid_benchmark PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8>)
  target_compile_definitions(windows_tensor_grid_benchmark PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
