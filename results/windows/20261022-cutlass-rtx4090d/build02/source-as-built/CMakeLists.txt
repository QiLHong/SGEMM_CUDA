cmake_minimum_required(VERSION 3.19)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 89 CACHE STRING "CUDA architectures")
endif()
project(CutlassRTX4090D LANGUAGES CXX CUDA)
find_package(CUDAToolkit 12.4 REQUIRED)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
set(CUTLASS_ROOT "" CACHE PATH "Unmodified NVIDIA CUTLASS v3.5.1 source root")
if(NOT EXISTS "${CUTLASS_ROOT}/include/cutlass/gemm/device/gemm.h")
  message(FATAL_ERROR "Set CUTLASS_ROOT to the reviewed CUTLASS source directory")
endif()
add_executable(windows_cutlass_4090d_benchmark benchmark.cu)
target_include_directories(windows_cutlass_4090d_benchmark PRIVATE "${CUTLASS_ROOT}/include")
target_link_libraries(windows_cutlass_4090d_benchmark PRIVATE CUDA::cublas)
target_compile_definitions(windows_cutlass_4090d_benchmark PRIVATE SGEMM_K10_MIN_BLOCKS=2)
target_compile_options(windows_cutlass_4090d_benchmark PRIVATE
  $<$<COMPILE_LANGUAGE:CUDA>:-O3;-lineinfo;-Xptxas=-v;--expt-relaxed-constexpr;--expt-extended-lambda>)
if(MSVC)
  target_compile_options(windows_cutlass_4090d_benchmark PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/utf-8>)
  target_compile_definitions(windows_cutlass_4090d_benchmark PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
